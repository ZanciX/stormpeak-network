# ============================================
# Citizens NPC Setup - StormPeak Hub
# Zaženi te ukaze v-game kot OP
# ============================================

# 1. Pojdi na lokacijo kjer želiš NPC (Hub spawn area)

# === NPC: Survival ===
# Postavi se kamor hoces Survival NPC
/npc create &6&lSurvival --type PLAYER
/npc skin Notch
/npc look
/trait actions
/npc command add -l left "server survival"
/npc command add -l right "server survival"

# === NPC: SkyPVP ===
/npc create &c&lSkyPVP --type PLAYER
/npc skin Dream  
/npc look
/trait actions
/npc command add -l left "server skypvp"
/npc command add -l right "server skypvp"

# === NPC: Skyblock ===
/npc create &b&lSkyblock --type PLAYER
/npc skin Steve
/npc look
/trait actions
/npc command add -l left "server skyblock"
/npc command add -l right "server skyblock"
